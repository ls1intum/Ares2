package %s.ares.api.aop.java;

import %1$s.ares.api.localization.Messages;

/** Holds the generated policy settings read by runtime advice. */
public final class JavaAOPTestCaseSettings {

    /** Serialises reflective reads and writes of this settings copy. */
    private static final Object SETTINGS_LOCK = new Object();

    /** Prevents creating instances of the settings holder. */
    private JavaAOPTestCaseSettings() {
        throw new SecurityException(Messages.localized("security.general.utility.initialization", "JavaAOPTestCaseSettings"));
    }

    /** Returns the same lock used by the packaged advice settings API. */
    public static Object getSettingsLock() {
        return SETTINGS_LOCK;
    }

    /**
     * The default temp directory, resolved to its real location, that trusted Ares
     * code fixed when the JVM started; {@code null} means nothing fixed it, and a
     * temp file without a named directory is then refused.
     */
    private static volatile String frozenDefaultTempDirectory = null;

    /**
     * Read by the shared runtime advice, which a test run ends through this field;
     * a generated set has no such end, so it stays {@code null} and every thread
     * is checked.
     */
    private static volatile Thread supervisionEndedOnThread = null;

    /**
     * Read by the shared runtime advice to lift every check for an unguarded phase;
     * a generated set has no such phase, so it stays {@code false}.
     */
    private static volatile boolean supervisionSuspended = false;
