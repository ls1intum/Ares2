/**
 * Architecture test case for the Java programming language using Wala and concrete product of the abstract factory design pattern.
 *
 * @author Sarp Sahinalp
 * @version 2.0.0
 * @see <a href="https://refactoring.guru/design-patterns/abstract-factory">Abstract Factory Design Pattern</a>
 * @since 2.0.0
 */
public class JavaWalaTestCase {

    /** The call graph of each supervised package, built once per run and shared by every check of this class. */
    private static final Map<String, CallGraph> CALL_GRAPHS = new ConcurrentHashMap<>();

    @Nonnull Set<PackagePermission> allowedPackages;
    @Nonnull JavaClasses javaClasses;
    @Nonnull CallGraph callGraph;

    /**
     * Tells whether the supervised package holds no compiled class yet. A check then has no code to judge and no
     * folder to build a call graph from, so it passes, as an ArchUnit rule over no class does.
     *
     * @param supervisedClasses the compiled classes of the supervised package
     * @return whether there are none
     */
    static boolean nothingToAnalyse(JavaClasses supervisedClasses) {
        return !supervisedClasses.iterator().hasNext();
    }

    /**
     * Returns the call graph of the compiled classes of the supervised package, built on first use.
     *
     * @param supervisedPackage the supervised package
     * @return the call graph
     */
    static CallGraph supervisedCallGraph(String supervisedPackage) {
        return CALL_GRAPHS.computeIfAbsent(supervisedPackage, scope -> {
            String classPath = JavaArchunitSupervisedClasses.supervisedClassPath(scope);
            return new CustomCallgraphBuilder(classPath).buildCallGraph(classPath);
        });
    }
